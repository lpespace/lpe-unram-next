lib/api.js · kirim header di setiap fetch publik
// URL API & slug merchant disimpan di .env
const API = process.env.NEXT_PUBLIC_API_URL;
const MERCHANT = process.env.NEXT_PUBLIC_MERCHANT_SLUG; // "rentallombok"

// Ambil daftar kendaraan (opsional filter: "mobil" / "motor")
export async function getVehicles(jenis) {
  const url = jenis
    ? `${API}/public/vehicles?jenis=${jenis}`
    : `${API}/public/vehicles`;
  const res = await fetch(url, {
    headers: { "X-Merchant-Slug": MERCHANT },
    next: { revalidate: 60 },   // cache 60 detik
  });
  if (!res.ok) throw new Error("Gagal memuat kendaraan");
  return res.json();
}

Standar Minimum Kelulusan

  • Dibangun dengan React + JavaScript (Next.js App Router)
  • Min. 4 route berjalan: /, /tentang, /kendaraan, /kendaraan/[slug]
  • Min. 2 komponen reusable (mis. VehicleCard, Button)
  • Data kendaraan dari API rental live (header X-Merchant-Slug) + filter Mobil/Motor
  • State loading / error / kosong tertangani
  • Halaman detail + tombol Booking via WhatsApp berfungsi
  • Foto teroptimasi (next/image) + metadata/SEO
  • Live di Vercel dengan env terpasang + repo GitHub rapi
  • Responsif (mobile → desktop)
